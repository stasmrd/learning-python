```python
class Developer:
    def __init__(self, name):
        self.name = name
        self.skills = ["Python", "Problem Solving"]
        self.goal = "Full Stack Developer"

    def learn(self, new_skill):
        self.skills.append(new_skill)
        print(f"🔥 Am învățat ceva nou: {new_skill}!")

    def __str__(self):
        return f"👨‍💻 {self.name} | Skills: {', '.join(self.skills)}"

me = Developer("Viitorul Programator")
me.learn("Python OOP")
print(me)
```
